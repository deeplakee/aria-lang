# aria 语言文法
#
# 产生式区只含文法规则；设计约束与解析要点见文末「说明」。
# 终结符：字面量用 "..." 包裹，[...] 为字符类，* / + / ? 为正则量词。
# 非终结符：PascalCase 或小写词组。

program        -> declaration*

declaration    -> funDecl | defDecl | varDecl | statement

funDecl        -> "fun" function
function       -> identifier params block

params         -> "(" paramsBody? ")"
paramsBody     -> plainParams ( "," defaultParams )? ( "," varargs )?
               | defaultParams ( "," varargs )?
               | varargs
plainParams    -> identifier ( "," identifier )*
defaultParams  -> defaultParam ( "," defaultParam )*
defaultParam   -> identifier "=" expression
varargs        -> "..." identifier

defDecl        -> "def" identifier ( ":" identifier )? "{" member* "}"
member         -> varDecl | funDecl | function

varDecl        -> "var" varBinding ( "," varBinding )* ";"
varBinding     -> varTarget ( "=" expression )?
varTarget      -> identifier | pattern

statement      -> printStmt | ifStmt | whileStmt | forStmt | forInStmt
               | breakStmt | continueStmt | returnStmt | importStmt
               | tryCatchStmt | throwStmt | matchStmt | block | exprStmt

printStmt      -> "print" expression ";"
ifStmt         -> "if" "(" expression ")" statement ( "else" statement )?
whileStmt      -> "while" "(" expression ")" statement
forStmt        -> "for" "(" ( varDecl | exprStmt | ";" ) expression? ";" expression? ")" statement
forInStmt      -> "for" "(" pattern "in" expression ")" statement
breakStmt      -> "break" ";"
continueStmt   -> "continue" ";"
returnStmt     -> "return" expression? ";"
importStmt     -> "import" string "as" identifier ";"
tryCatchStmt   -> "try" block ( "catch" "(" identifier ")" block )?
throwStmt      -> "throw" expression ";"
matchStmt      -> "match" "(" expression ")" "{" matchArm+ "}"
matchArm       -> matchPattern "=>" statement
matchPattern   -> expression | "_"
block          -> "{" declaration* "}"
exprStmt       -> expression ";"

expression     -> assignment

assignment     -> logic_or ( ( "=" | "+=" | "-=" | "*=" | "/=" | "%=" ) assignment )?
               | listPattern "=" assignment

logic_or       -> logic_and ( "||" logic_and )*
logic_and      -> equality ( "&&" equality )*
equality       -> comparison ( ( "!=" | "==" | "!==" | "===" ) comparison )*
comparison     -> range ( ( ">" | ">=" | "<" | "<=" ) range )*
range          -> term ( ( ".." | "..." ) term )?
term           -> factor ( ( "-" | "+" ) factor )*
factor         -> unary ( ( "/" | "*" | "%" ) unary )*
unary          -> ( "-" | "!" | "++" | "--" ) unary | value
value          -> primary ( args | "." identifier | "[" expression "]" )*
args           -> "(" ( expression ( "," expression )* )? ")"

primary        -> number | string | identifier | "nil" | "false" | "true"
               | "this" | "super" | parenExpr | listExpr | mapExpr | ifExpr | lambdaExpr | matchExpr

parenExpr      -> "(" expression ")"
listExpr       -> "[" ( expression ( "," expression )* )? "]"
mapExpr        -> "{" ( pair ( "," pair )* )? "}"
pair           -> expression ":" expression
ifExpr         -> "if" "(" expression ")" "{" expression "}" "else" "{" expression "}"
lambdaExpr     -> "fun" params block
matchExpr      -> "match" "(" expression ")" "{" matchExprArm+ "}"
matchExprArm   -> matchPattern "=>" expression

# --- 解构模式 ---
# pattern 含 "_"：占位符，匹配/忽略该位置不绑定（用于跳过不关心的元素）。
pattern        -> identifier | "_" | listPattern
listPattern    -> "[" ( pattern ( "," pattern )* ( "," restPattern )? | restPattern )? "]"
restPattern    -> "..." identifier

# =============================================================================
# 词法规则（lexer 层，非上下文无关文法）
# =============================================================================

number         -> int | float
int            -> decimal | bin | oct | hex
decimal        -> [0-9] ( [0-9] | _ [0-9] )*
bin            -> "0" ("b"|"B") [01] ( [01] | _ [01] )*
oct            -> "0" ("o"|"O") [0-7] ( [0-7] | _ [0-7] )*
hex            -> "0" ("x"|"X") [0-9a-fA-F] ( [0-9a-fA-F] | _ [0-9a-fA-F] )*
float          ->   [0-9] ( [0-9] | _ [0-9] )* "." [0-9] ( [0-9] | _ [0-9] )* exp?    # 1.5 / 1.5e2
                 | [0-9] ( [0-9] | _ [0-9] )* exp        # 1e2 无小数点须带指数
exp            -> ("e"|"E") ("+"|"-")? [0-9] ( [0-9] | _ [0-9] )*

string         -> plainString
plainString    -> "\"" char* "\"" | "'" char* "'"
char           -> 非(引号 | 反斜杠 | 换行) 字符 | escape
escape         -> "\\" ( "\"" | "'" | "\\" | "n" | "t" | "r" | "0" | "u{" hex+ "}" )

identifier     -> id_start id_continue*

comment        -> "//" [^\n]* | "#" [^\n]*

# =============================================================================
# 说明
# =============================================================================
#
# 总则：本文法描述 parse tree 形态，不编码语义约束。下列语义检查推迟到语义分析 /
# 字节码生成阶段：左值/右值合法性（assignment 左操作数须为 identifier / obj.field /
# obj[index]）、super 须为 super.method(...) 形态、break/continue 须在循环内、
# return 须在函数内、try 须有 catch、默认值表达式不得引用同函数参数。
#
# 复合赋值求值不变式：E1 op= E2 等价于 E1 = E1 op E2，但左值 E1 只求值一次（C 标准
# 对 E1 += E2 的定义，非字面展开的两次）。对带副作用左值关键：arr[f()] += 1 中 f()
# 只调一次，getObj().field += 1 中 getObj() 只调一次。字节码生成须走单次求值 lowering--
# Identifier 左值重解析一遍（名字解析无副作用，locator-once 天然成立），Field/Index 左值
# 经栈上 DUP/DUP2 保留 locator（见 compound-assignment-lowering.md），不得将带副作用的
# LHS 子树求值两遍。普通 = 无此约束。
#
# 表达式与语句的边界：if/fun/while/for 等既是语句关键字，又能出现在表达式位置
# （ifExpr、lambdaExpr）。parser 按调用上下文调度--解析 statement 时走语句产生式，
# 解析 expression 时走表达式产生式，无文法层冲突。aria 无逗号表达式，故 "a, b" 仅
# 出现在解构模式中。
#
# 运算符优先级（从低到高）：赋值 < || < && < 相等 < 比较 < 区间 < 加减 < 乘除模 < 一元
# (含前置 ++/--/!/-) < 后缀调用/取字段/取下标。同层多为左结合；一元右结合；
# 赋值右结合。相等层含 ==/!=/===/!==（同优先级、左结合）：== 内容相等(value_equal，
# Int/F64 跨类型 IEEE 数值、Obj 调 Object::equals 虚函数)、=== 严格相等(value_identical，
# 类型严格、f64 按位、Obj 指针)；!= / !== 分别为 == / === 的取反（NOT_EQUAL /
# STRICT_NOT_EQUAL）。逻辑运算 (||/&&) 短路求值。前置 ++/-- 作用于后续 unary（自增/自减），
# 左值合法性留语义阶段。区间 a..b（含上界）/ a...b（不含上界）产生 range 对象，非结合
# （a..b..c 非法）；... 兼作 rest/varargs 前缀，按位置消歧。
#
# 分号与空语句：除 varDecl/print/return/break/continue/throw/import 外，多数语句通过
# exprStmt 要求结尾 ";"。forStmt 的 init 部分允许 ";" 表示空 init。单独 ";"（空语句）
# 当前未列入 statement，如需支持再加 emptyStmt。
#
# 上下文消歧（parser 前瞻，文法层无冲突）：
#   - for vs forIn：均以 "for" "(" 开头。"(" 后为 ";" -> forStmt 空 init；为 "var" ->
#     forStmt 的 varDecl init；为 <pattern> "in"（identifier/"_" 紧跟 in，或 [...] 后跟 in）
#     -> forIn（in 非表达式运算符，故 <pattern> in 唯一标识 forIn）；否则按 forStmt 的
#     exprStmt init 解析（含裸表达式 init 如 i = 0）。forIn 靠 "in" 确认。
#   - if 语句 vs if 表达式：statement 位置遇 "if" -> ifStmt；expression 位置 -> ifExpr。
#   - match 语句 vs match 表达式：statement 位置遇 "match" -> matchStmt；expression
#     位置 -> matchExpr。
#   - fun 声明 vs lambda：statement 位置遇 "fun" + identifier -> funDecl；expression
#     位置遇 "fun" + "(" -> lambdaExpr。
#   - def 体内成员分派：首 token 为 "var" -> 静态变量；"fun" + identifier -> 静态方法；
#     裸 identifier -> 实例方法（identifier params block）。"fun" 后非标识符（lambda）不得
#     作成员，报错。
#   - "{" 两用：block（statement 位置）、mapExpr（expression 位置）。按调用上下文区分。
#
# 解构：pattern 出现在 var 声明的 varTarget 与 forIn 的目标；解构赋值以 listPattern
# 为左侧目标（assignment 的第二分支）。listPattern 与 listExpr 均以 "[" 起头。
#   - listPattern 映射为下标访问：位置 i 绑 list[i]。多余忽略、不足按越界报错
#     （复用下标越界检查，无需专门长度校验）。"_" 占位忽略该位置（如 [a, _, c]）。
#   - rest 可选：...rest 收集 list[i..] 为新 list；不写则默认忽略剩余。rest 必须绑名
#     （identifier），不接受 "_"--因 "..._" 与"不写 rest"等价，冗余非法。rest 须在末尾。
#
# 默认参数：paramsBody 顺序固定 plainParams -> defaultParams -> varargs，文法强制默认
# 参数后不得有无默认参数。默认值在调用且该参数未传时求值，可与 varargs 共存（按位置
# 填充，未传走默认，剩余进 varargs）。
#
# map 键为任意 Value（HashMap<Value,Value>）：mapExpr 的键是任意 expression（运行时求值）。
#
# forIn 遍历协议：被遍历对象产出 iterator（has_next/next，next 返回单个 Value）。
# list->元素、map->[key,value] 元组、string->字符。目标 pattern 绑 next() 的值：
# identifier 直接绑、listPattern 按位置解构（[k,v] 绑 next()[0]/[1]，复用下标语义）；
# "_" 占位忽略。enumerate 等内建函数留待以后。
#
# match：多分支按值匹配（if-else 链的语法糖，当前不支持解构模式）。从上到下取第一
# 个匹配分支执行/取值。matchPattern：expression（运行时求值后与 subject 比较相等，
# 非绑定）或 "_"（通配兜底）。无匹配且无 "_" -> 运行时错误。可作语句（matchStmt，
# 分支为单条 statement--多语句需用 block 包裹（block 亦属 statement））与表达式（matchExpr，分支为 expression，无花括号）。
# "=>" 为新引入符号。
#
# try/finally 裁撤记录（2026-09）：finally 子句已从文法移除（tryCatchStmt 仅存可选 catch，
# 语义要求 try 须有 catch），`finally` 回归普通标识符。理由：finally 的存在价值是资源善后，
# 而语言层尚无制造 OS 资源的内建（builtins 仅 type/len/str/assert）；参照实现 Lua/Wren 均
# 无 finally；其控制流语义是剩余异常机制中最贵的一块。曾定稿「finally 体内控制流不得离开
# finally」的编译期禁止方案，细则存档于 exception-implementation-pitfalls.md「M3b finally
# 裁撤记录」。善后后继为 defer（Go/Zig 式：defer 注册善后表达式，函数正常退出与异常
# unwind 时 LIFO 执行）；defer 已降为可选后续（优先级最低，其他功能完成后另定，2026-09
# 决定不再绑定 M4），细节重启设计时定。
#
# def / 静态成员 / 实例方法：defDecl 体内三种成员（member -> varDecl | funDecl | function）--
#   - varDecl（var ...;）-> 静态变量，属类本身（Foo.x 读写）。
#   - funDecl（fun name() {...}，带 fun 关键字）-> 静态方法，无 this 绑定，经 ClassName.x 访问静态成员。
#   - function（name() {...}，无 fun 关键字）-> 实例方法，有 this 绑定（原 class 体方法形式）。
#   体内裸名解析顺序：局部变量 -> upvalue -> 模块全局（类静态不在裸名作用域，经 ClassName.x）。Foo(args) 实例化；实例方法经
#   superclass（":" identifier）链继承，super.m(...) 仍可用。外部访问：Foo.x 取/设静态成员；
#   foo.x 先查实例自身属性，未命中再回退静态成员（实例属性遮蔽同名静态成员）。静态/实例方法
#   区分、访问顺序与遮蔽语义属编译期/VM 范畴，文法不编码（仅约束 member 的三种起首写法）。
#   静态成员继承（编译期/VM 语义，文法不编码）：静态变量与静态方法同实例方法一样经
#   superclass 链继承。子类未重声明时，Sub.x / Sub.m() 沿链命中父类槽，与 Super.x / Super.m()
#   共享同一存储（共享槽）；子类重声明则开新槽遮蔽父类（两份并存，静态绑定，非覆写）。类静态
#   经 ClassName.x 限定访问（运行期走 super），不在裸名作用域（见下文作用域模型）；方法查找
#   靠内联缓存 / bound method 缓存加速（闭包不可变，可缓存）。
#   Object 根类（编译期/VM 语义，文法不编码）：所有用户定义的类隐式继承自内置 Object 根类--
#   `def Foo`（无显式父类）等价于 `def Foo : Object`，`def Foo : Bar` 经 Bar 链最终到达 Object。
#   Object 是唯一 superclass 为空的类，提供最小通用方法（如 init 的默认 no-op 实现）；链式查找
#   （静态/实例方法/foo.x 回退）统一终止于 Object（裸名走模块全局，不在此列）。原始值（nil/bool/数字，非 Obj）
#   不在 Object 层次内；内置 Obj 类型（string/list/map 等）暂亦不在内；纳入为 uniform OOP 目标方向（尚未实现）。
#   作用域模型（编译期/VM 语义，文法不编码，Python/JS 路子）：
#   - 模块 = 命名空间（非类）：文件是一个模块，顶层 var/fun/class 是模块级绑定（模块全局），
#     不是某个类的静态；顶层可执行语句编进模块体，导入时跑一次。模块存于 VM 内部模块表。
#   - ObjClass 两指针：super（继承）、meta（元类，未来）。两根：
#     Object（super=nil，继承终止）、meta（meta=meta 自指，未来）。
#   - 裸名解析：局部（LOAD_LOCAL）-> upvalue（LOAD_UPVALUE）-> 模块全局（LOAD_GLOBAL，
#     运行期查模块全局表）。不走 enclosing+super，不查类静态--类静态不在裸名作用域。未命中
#     报错。裸名只看局部、upvalue、模块全局。
#   - 裸名赋值（写路径，与读对称）：走同路径（STORE_LOCAL / STORE_UPVALUE / STORE_GLOBAL
#     查模块全局），未命中报错（必须先 var 声明）。var 声明：函数体=局部、模块顶层=模块全局、
#     类体=静态（经 ClassName.x 访问，不进裸名）。
#   - 类静态访问（限定，运行期）：ClassName.x（沿 super 查，含继承共享槽）、this.x/obj.x
#     （先查实例字段，未命中回退类静态）。裸名看不到类静态。
#   - 嵌套类（全路径）：class B 在 class A 内 -> B 是 A 的静态 A.B，经 A.B 全路径引用，无
#     enclosing 链、无裸名。模块顶层类是模块全局，裸名可访；嵌套类须全路径。
#   - ObjFn 持 defining class：用于 super（LOAD_SUPER_METHOD）。无 enclosing 走链。
#   - this（实例，小写）为捕获 upvalue（arrow-function 语义）：this 是实例方法槽 1；嵌套函数
#     引用 this 时，沿外围函数帧找最近的实例方法，捕获其槽 1 为 upvalue（this -> LOAD_UPVALUE，
#     .x -> LOAD_FIELD）。链上无实例方法（静态方法 / 顶层 / 只嵌在静态方法里）则 this 不可用，
#     编译期报错。静态方法本身无 this。
#   - super 边界：super.m() 仅实例方法可用（槽 1 this 调被覆写实例方法）；父类静态直接
#     ClassName.x，不支持 super.x。静态方法无 this，super 非法。
#   - 静态初始化（eager）：静态初始化器类定义时求值，非首次访问；class A{var x=B();} 要求
#     B 先于 A 定义。
#   - 动态加静态：不支持 monkey-patch，Foo.newStatic=v（新名）报错；静态必须 var 声明（实例
#     字段动态，有意不对称）。
#   - 无 This 关键字：自引用写类名（Foo.x）。
#   - 模块导入：文件是模块（非类）。启动定源根列表（标准库 lib + 入口文件目录，环境变量留
#     后续）；导入只用字符串路径，不支持裸名：import "lib/utils" as Utils（绝对，从源根搜）、import
#     "./utils" as U（相对当前文件）、import "lib" as Lib（目录包）；强制 as alias--绑模块到当前作用域
#     （模块顶层=模块全局、函数体/块内=局部，IMPORT 压模块值于栈顶，绑定经 DEF_GLOBAL（顶层）
#     或值填槽 + mark_initialized（嵌套）按作用域走），不解析路径算模块名。相对导入不得越出
#     当前文件所属源根（../ 爬到源根之上报错）。目录 = 包（路径结构，非嵌套类）。循环导入
#     返回半初始化对象，裸名不自动导入。符号链接按规范路径判定、重叠源根按列表首次命中。
#     标记文件作源根是未来 aria run 特性，暂不支持。
#
# 词法要点：
#   - int 为 i48（[-2^47, 2^47)），float 为 f64；溢出/精度语义阶段处理。
#   - number 按是否含小数点或指数区分：含点或 e/E 指数 -> float（1.5 / 1.5e2 /
#     1e2 / 1e-1）；无点且无指数 -> int（10 / 0xFF）。科学计数法（e/E）一律作
#     float，不再走 int。
#   - float 的小数点两侧必须各有至少一位数字：禁止 .5 / 5. / 1.e2 / .5e2 这类
#     不完整形式（.5 不再启动数字扫描，. 作 Dot token；5. 作 int 5 + Dot）。
#   - _ 仅出现在数字之间（首尾、紧邻前缀/点/e 非法）；数字字面量不跨行。
#   - 字符串首尾引号须一致、不跨行、裸换行非法；转义 \" \' \\ \n \t \r \0 \u{...}，
#     未识别转义非法（无 \x）。无插值字符串（f"..." 不作特殊处理，f 视为普通 identifier）。
#   - identifier：id_start（_ 或 Unicode 字母）+ id_continue*（并 Unicode 数字），
#     复用 utf8.hpp 的 is_id_start/is_id_continue（区间近似，非完整 UCD）；大小写敏感；
#     最长匹配；关键字优先于 identifier。
#   - 注释 // 或 # 到行尾，lexer 丢弃，不产生 token；无块注释。
#
# 关键字：fun def var if else while for in break continue return import as try catch
# throw print nil true false this super match（... 仅作 varargs/rest
# 前缀，非独立关键字；_ 为占位/通配符，在 matchPattern 与 pattern 中识别，不绑定）。
